name: Build IPA

on:
  workflow_dispatch:
  push:
    branches: [ main, master ]

jobs:
  build:
    runs-on: macos-14
    timeout-minutes: 30

    steps:
      - name: Checkout
        uses: actions/checkout@v4

      - name: Install xcodegen
        run: brew install xcodegen

      - name: Generate Xcode project
        run: xcodegen generate

      - name: Build (unsigned)
        run: |
          set -o pipefail
          xcodebuild \
            -project HaiTac.xcodeproj \
            -scheme HaiTac \
            -configuration Release \
            -sdk iphoneos \
            -derivedDataPath build \
            ARCHS=arm64 \
            ONLY_ACTIVE_ARCH=NO \
            CODE_SIGNING_ALLOWED=NO \
            CODE_SIGNING_REQUIRED=NO \
            CODE_SIGN_IDENTITY="" \
            build

      - name: Package .ipa
        run: |
          mkdir -p Payload
          cp -R build/Build/Products/Release-iphoneos/HaiTac.app Payload/
          zip -qry HaiTac.ipa Payload
          ls -lh HaiTac.ipa

      - name: Upload IPA
        uses: actions/upload-artifact@v4
        with:
          name: HaiTac-ipa
          path: HaiTac.ipa
          retention-days: 30
